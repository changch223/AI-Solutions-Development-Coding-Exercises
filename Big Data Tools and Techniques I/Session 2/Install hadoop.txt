*************************************************************
**1.	Create a Resource Group
*************************************************************
az group create --name trainingrg --location eastus --tags Name=Soosan Environment=Training


*************************************************************
**2.	Create SSH Key and use the same to Create Virtual machine
*************************************************************
ssh-keygen -m PEM -t rsa -b 2048
**press enter for default filename/provide no passphrase for clarity

az vm create --resource-group trainingrg --name MyUbuntu --image Ubuntu2204  --admin-username azuresoosan --size Standard_F8s --ssh-key-value ~/.ssh/id_rsa.pub


*************************************************************
**3.	Install Web Server
*************************************************************
az vm run-command invoke -g trainingrg -n MyUbuntu  --command-id RunShellScript  --scripts "sudo apt-get update && sudo apt-get install -y nginx"

*************************************************************
**4.	Open port 80 for web traffic
*************************************************************
az vm open-port --port 80 --resource-group trainingrg --name MyUbuntu

*************************************************************
**12. Configure hadoop
*************************************************************
-------------------------------------------------------------
sudo nano "$HADOOP_HOME/etc/hadoop/core-site.xml"

**append following between configuration bucket
	
 <property>
    <name>fs.defaultFS</name>
    <value>hdfs://localhost:9000</value>
  </property>

**end of append , Ctrl+S AND Ctrl+X to save and exit


sudo nano "$HADOOP_HOME/etc/hadoop/hdfs-site.xml"

**append following between configuration bucket

  <property>
    <name>dfs.replication</name>
    <value>1</value>
  </property>

**end of append , Ctrl+S AND Ctrl+X to save and exit

***now setup ssh key for localhost
	
ssh-keygen
**just press enter for everything default
cat $HOME/.ssh/id_rsa.pub >> $HOME/.ssh/authorized_keys
chmod 600 $HOME/.ssh/authorized_keys

ssh localhost
exit
