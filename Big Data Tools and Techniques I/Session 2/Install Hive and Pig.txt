Install HIVE
***********************************************************************************************************

sudo apt-get update

wget http://archive.apache.org/dist/hive/hive-3.1.2/apache-hive-3.1.2-bin.tar.gz
tar -xzf  apache-hive-3.1.2-bin.tar.gz


sudo mv apache-hive-3.1.2-bin /usr/local/hive
ls /usr/local


nano .bashrc

**append following between configuration bucket


#Hadoop Path
export HADOOP_HOME=/usr/local/hadoop
export PATH=$PATH:$HADOOP_HOME/bin
export PATH=$PATH:$HADOOP_HOME/sbin
export HADOOP_MAPRED_HOME=${HADOOP_HOME}
export HADOOP_COMMON_HOME=${HADOOP_HOME}
export HADOOP_HDFS_HOME=${HADOOP_HOME}
export YARN_HOME=${HADOOP_HOME}

#Hive configurations
export HIVE_HOME=/usr/local/hive
export PATH=$PATH:$HIVE_HOME/bin
export HIVE_CONF_DIR=$HIVE_HOME/conf
export CLASSPATH=$CLASSPATH:$HADOOP_HOME/lib/*:.
export CLASSPATH=$CLASSPATH:$HIVE_HOME/lib/*:.


**end of append, ctr+s and ctrl+x to save and exit

source .bashrc


cd $HIVE_HOME
cd conf
ls
sudo cp hive-env.sh.template hive-env.sh
sudo nano hive-env.sh



***search (f6) HADOOP_HOME AND remove the # and replace following
HADOOP_HOME=/usr/local/hadoop

****append following into first config

  <property>
    <name>system:java.io.tmpdir</name>
    <value>/tmp/hive/java</value>
  </property>
  <property>
    <name>system:user.name</name>
    <value>${user.name}</value>
  </property>
** end of append m ctrl+s and ctrl+x to save and exit



hadoop fs -mkdir -p /user/hive/warehouse
